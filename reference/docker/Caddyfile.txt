:8000 {
    # فحص أمني: منع أي محاولة لتجاوز المسار
    # استخدام تعبير نمطي للبحث عن .. بشكل صريح
    @path_traversal path_regexp \.\.
    handle @path_traversal {
        respond "Path Traversal Attempt Denied" 403
    }

    # نطابق المسارات التي تبدأ بمعرف المستخدم ثم أي مسار بعده
    # مثال: /12345/bot.php أو /12345/subdir/another.php
    @bot_request path_regexp bot ^/(\d+)/(.+)$
 
    handle @bot_request {
        # تعريف متغير صريح لمسار المستخدم
        vars root /app/user_bots/{re.bot.1}
        # استخدام المتغير لتعيين المجلد الجذري
        root * {http.vars.root}
        rewrite * /{re.bot.2}
 
        # استخدام المتغير في كل مكان لضمان الثبات
        php_fastcgi 127.0.0.1:9000 {
            env PHP_ADMIN_VALUE "open_basedir={http.vars.root}:/tmp/ \n auto_prepend_file={http.vars.root}/host_bootstrap.php"
        }
    }

    # التعامل مع أي طلبات أخرى غير صالحة
    handle {
        respond "Invalid Request" 404
    }
}
